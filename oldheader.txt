#ifndef LINKEDLIST_H
#define LINKEDLIST_H

#include <iostream>
#include <fstream>
using namespace std;

template <typename T>
class linkedList {
private:
        class node {
        public:
            node *next;
            T data;

            node(T d);
        };

        node *head;

public:
        // constructor
        linkedList::singleLL();

        // // destructor
        // ~singleLL();

        // print each element in LL 
        void linkedList::print();

        // add to end of LL
        T linkedList::push_back(T d);

        // takes 2 sorted lists and merges them, O(n)
        void linkedList::merge(node l, node r);

        // open file, enter each word in ll element
        void linkedList::loadFromFile(string w);
    
        // write each item in ll to output file
        void linkedList::writeToFile(string w);

};


#endif
